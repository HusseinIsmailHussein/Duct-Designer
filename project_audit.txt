class ItemsDatabaseHelper {
  ItemsDatabaseHelper._();

  static final ItemsDatabaseHelper db = ItemsDatabaseHelper._();

  final database = DatabaseRepository().database;

  Future<List<Item>> getAllItems() async {
    var db = await database;
    List<Map<String, dynamic>> maps = await db.query(tableItem);
    List<Item> list =
        maps.isNotEmpty ? maps.map((item) => Item.fromMap(item)).toList() : [];
    return list;
  }

  Future<Item> getItem(int itemId) async {
    var db = await database;
    List<Map<String, dynamic>> maps = await db.rawQuery('''
    SELECT * FROM $tableItem 
    WHERE ${ItemFields.id} = $itemId
    ''');
    Item item = Item.fromMap(maps.first);
    return item;
  }

  Future<List<Item>> getItemsBySystem(System system) async {
    var db = await database;
    List<Map<String, dynamic>> maps = await db.rawQuery('''
    SELECT * FROM $tableItem 
    WHERE ${ItemFields.systemId} = ${system.id}
    ''');
    List<Item> list =
        maps.isNotEmpty ? maps.map((item) => Item.fromMap(item)).toList() : [];
    return list;
  }

  insert(Item model) async {
    var db = await database;
    await db.insert(tableItem, model.toMap());
  }

  update(Item item) async {
    var db = await database;
    await db.update(tableItem, item.toMap(),
        where: '${ItemFields.id} =?', whereArgs: [item.id]);
  }

  delete(Item model) async {

import 'package:math_parser/math_parser.dart';

import '../model/duct_model.dart';
import '../model/item_model.dart';

class DuctFunctions {
  Set<String> getVariablesSet(DuctModel duct, Item item) {
    Set<String> variablesSet = {};
    for (var i = 0; i < duct.variables.length; i++) {
      variablesSet.add(duct.variables.elementAt(i).name);
    }
    return variablesSet;
  }

  Map<String, num> getVariablesMap(DuctModel duct, Item item) {
    Map<String, num> variablesMap = {};
    for (var i = 0; i < duct.variables.length; i++) {
      variablesMap[duct.variables.elementAt(i).name] = item.variables[i];
    }
    return variablesMap;
  }

  List<String> getErrors(DuctModel duct, Item item) {
    if (duct.tests != null) {
      List<String> list = [];
      for (var i = 0; i < duct.tests!.length; i++) {
        if (MathNodeExpression.fromString(duct.tests![i][0],
                    variableNames: getVariablesSet(duct, item))
                .calc(MathVariableValues(getVariablesMap(duct, item)))
                .toDouble() <
            0) {
          list.add(duct.tests![i][1]);
        }
      }
      return list;
    } else {
      return [];
    }
  }

  double getTotalArea(DuctModel duct, Item item, List<String> errors) {
    if (errors.isEmpty) {
      return MathNodeExpression.fromString(duct.expression,
              variableNames: getVariablesSet(duct, item))
          .calc(MathVariableValues(getVariablesMap(duct, item)))
          .toDouble();
    } else {
      return 0;
    }
  }
}

      {String? gauge,
      required String material,
      required int id,
      required double thickness,
      required String unit}) {
    double density;
    if (material == Materials.aluminum) {
      density = densityAL;
    } else {
      density = densityGI;
    }
    return MaterialDataModel(
        gauge: gauge,
        id: id,
        material: material,
        density: density * thickness,
        thickness: thickness,
        unit: unit);
  }

      {required String material,

        thickness: thickness,

import 'package:path/path.dart';
import 'package:sqflite/sqflite.dart';
import '../../! core/model/units/material.dart';
import '../constant/constance.dart';
import '../../! core/model/item_model.dart';
import '../../! core/model/project_model.dart';
import '../../! core/model/system_model.dart';
import '../constant/material_list.dart';

class DatabaseRepository {
  DatabaseRepository();

  static final DatabaseRepository db = DatabaseRepository();

  static Database? _database;

  get database async {
    if (_database != null) {
      return _database;
    }
    _database = await initDb();
    return _database;
  }

  initDb() async {
    String path = join(await getDatabasesPath(), 'database.db');
    const String idType = 'INTEGER PRIMARY KEY AUTOINCREMENT';
    const String intType = 'INTEGER NOT NULL';
    const String textType = 'TEXT NOT NULL';
    const String doubleType = 'REAL NOT NULL';

    return await openDatabase(path, version: 1,
        onCreate: (Database db, int version) async {
      await db.execute(''' CREATE TABLE $tableProject (
        ${ProjectFields.id} $idType,
        ${ProjectFields.name} $textType,
        ${ProjectFields.dateCreated} $textType,
        ${ProjectFields.dateModified} $textType,
        ${ProjectFields.totalWeight} $doubleType,
        ${ProjectFields.totalArea} $doubleType,
        ${ProjectFields.numberOfUnits} $intType,
        ${ProjectFields.numberOfItems} $intType) ''');

      await db.execute(''' CREATE TABLE $tableSystem (
        ${SystemFields.id} $idType,
        ${SystemFields.projectId} $intType,
        ${SystemFields.name} $textType,
        ${SystemFields.quantity} $intType,
        ${SystemFields.dateCreated} $textType,
        ${SystemFields.dateModified} $textType,
        ${SystemFields.totalWeight} $doubleType,
        ${SystemFields.totalArea} $doubleType,
        ${SystemFields.numberOfItems} $intType) ''');

      await db.execute(''' CREATE TABLE $tableItem (
        ${ItemFields.id} $idType,
        ${ItemFields.ductId} $intType,
        ${SystemFields.projectId} $intType,
        ${ItemFields.systemId} $intType,
        ${ItemFields.quantity} $intType,
        ${ItemFields.dateCreated} $textType,
        ${ItemFields.variables} $textType,
        ${ItemFields.dateModified} $textType,
        ${ItemFields.totalWeight} $doubleType,
        ${ItemFields.totalArea} $doubleType)''');

      await db.execute(''' CREATE TABLE $tableMaterials (
        ${MaterialFields.id} $intType,
        ${MaterialFields.material} $textType,
        ${MaterialFields.gauge} $textType,
        ${MaterialFields.density} $doubleType,
        ${MaterialFields.thickness} $doubleType,
        ${MaterialFields.unit} $textType)''');

      for (var i = 0; i < standardMaterialList.length; i++) {
        await db.insert(tableMaterials, standardMaterialList[i].toMap());
        print(standardMaterialList[i].id);
      }
    });
  }
}

  void updateProjectDataDB() async {
    if (indexP != null) {
      double area = 0;
      double weight = 0;
      int numberOfItems = 0;
      int numberOfUnits = 0;
      List<System> systems = _projectsList[indexP!].systems;
      for (var index = 0; index < systems.length; index++) {
        area = area + systems[index].totalArea * systems[index].quantity;
        weight = weight + systems[index].totalWeight * systems[index].quantity;
        numberOfItems = numberOfItems + systems[index].numberOfItems;
        numberOfUnits = numberOfUnits + systems[index].quantity;
      }
      _projectsList[indexP!].totalArea = area;
      _projectsList[indexP!].totalWeight = weight;
      _projectsList[indexP!].numberOfItems = numberOfItems;
      _projectsList[indexP!].numberOfUnits = numberOfUnits;
    }
  }

  void updateSystemDataDB() async {
    if (indexP != null && indexS != null) {
      double area = 0;
      double weight = 0;
      int numberOfItems = 0;
      List<Item> items = _projectsList[indexP!].systems[indexS!].items;
      for (var index = 0; index < items.length; index++) {
        area = area + items[index].totalArea * items[index].quantity;
        weight = weight + items[index].totalWeight * items[index].quantity;
        numberOfItems = numberOfItems + items[index].quantity;
      }
      _projectsList[indexP!].systems[indexS!].totalArea = area;
      _projectsList[indexP!].systems[indexS!].totalWeight = weight;
      _projectsList[indexP!].systems[indexS!].numberOfItems = numberOfItems;
    }
  }

class MaterialSettingsController extends GetxController {
  var materialDBHelper = MaterialsDatabaseHelper.db;
  List<MaterialDataModel> _materialList = [];
  List<MaterialDataModel> get materialList => _materialList;

  String _material = Materials.galvanizedSteel;
  String get material => _material;
  set material(String material) {
    _material = material;
    update();
  }

  String _thicknessUnit = DimensionName.metric;
  String get thicknessUnit => _thicknessUnit;
  set thicknessUnit(String thicknessUnit) {
    _thicknessUnit = thicknessUnit;
    update();
  }

  MaterialSettingsController() {
    getAllMaterial();
    update();
  }

  getAllMaterial() async {
    _materialList = await materialDBHelper.getAllItems();
    update();
  }

  set materialList(List<MaterialDataModel> materialList) {
    _materialList = materialList;
    update();
  }

  updateMaterial(MaterialDataModel material) {
    MaterialDataModel selected =
        _materialList.firstWhere((element) => element.id == material.id);
    int index = _materialList.indexOf(selected);
    _materialList[index] = material;
    update();
  }

  resetMaterialSetting() {
    _materialList =
        standardMaterialList.map((e) => MaterialDataModel.clone(e)).toList();
    update();
  }

  saveMaterialSetting() {
    for (var i = 0; i < _materialList.length; i++) {
      materialDBHelper.update(_materialList[i]);
    }
  }
}

class DataTableItems extends StatelessWidget {
  DataTableItems({super.key});
  ProjectsController controller = Get.find();
  ItemController itemController = Get.find();
  DataTableViewModel dataTableViewModel = Get.find();

  final columns = [
    'Id',
    'Items Name',
    'Qty',
    'Area',
    'Weight',
    'Created Date',
    'Modified Date'
  ];

  @override
  Widget build(BuildContext context) {
    return DataTable(
      columnSpacing: 5,
      sortAscending: controller.isAscending,
      sortColumnIndex: controller.sortColumnIndex,
      columns: getColumns(columns, controller),
      rows: getRows(controller),
    );
  }

  List<DataColumn> getColumns(
          List<String> columns, ProjectsController controller) =>
      columns
          .map(
            (column) => DataColumn(
              label: Text(column, style: TextStyles.tableHeader),
              onSort: (columnIndex, ascending) {
                List<Item> items = controller.selectedSystem!.items;
                if (columnIndex == 0) {
                  items.sort((item1, item2) =>
                      compareString(ascending, item1.id ?? 0, item2.id ?? 0));
                } else if (columnIndex == 1) {
                  items.sort((item1, item2) =>
                      compareString(ascending, item1.name, item2.name));
                } else if (columnIndex == 2) {
                  items.sort((item1, item2) =>
                      compareString(ascending, item1.quantity, item2.quantity));
                } else if (columnIndex == 3) {
                  items.sort((item1, item2) => compareString(
                      ascending, item1.totalArea, item2.totalArea));
                } else if (columnIndex == 4) {
                  items.sort((item1, item2) => compareString(
                      ascending, item1.totalWeight, item2.totalWeight));
                } else if (columnIndex == 5) {
                  items.sort((item1, item2) => compareString(
                      ascending, item1.dateCreated, item2.dateCreated));
                } else if (columnIndex == 6) {
                  items.sort((item1, item2) => compareString(
                      ascending, item1.dateModified, item2.dateModified));
                }

                controller.selectedSystem?.items = items;
                controller.sortColumnIndex = columnIndex;
                controller.isAscending = ascending;
              },
            ),
          )
          .toList();

  int compareString(bool ascending, dynamic value1, dynamic value2) =>
      ascending ? value1.compareTo(value2) : value2.compareTo(value1);

  List<DataRow> getRows(ProjectsController controller) {
    Item? selectedItem = controller.selectedItem;
    List<Item> items = controller.selectedSystem!.items;
    SelectedSettingsModel settings = controller.selectedSettings;

    return items
        .map(
          (item) => DataRow(
            cells: [
              DataCell(Text('${items.indexOf(item) + 1}',
                  style: TextStyles.tableBody)),
              DataCell(Text(item.name, style: TextStyles.tableBody)),
              DataCell(Center(
                  child:
                      Text('${item.quantity}', style: TextStyles.tableBody))),
              DataCell(Center(
                  child: Text(
                      '${toUnit(settings.area, item.totalArea).toPrecision(1)} ${settings.area.name}',
                      style: TextStyles.tableBody))),
              DataCell(Center(
                  child: Text(
                      '${toUnit(settings.weight, item.totalWeight).toPrecision(1)} ${settings.weight.name}',
                      style: TextStyles.tableBody))),
              DataCell(Center(
                  child: Text(DateFormat('yyyy-MM-dd').format(item.dateCreated),
                      style: TextStyles.tableBody))),
              DataCell(Center(
                  child: Text(
                      DateFormat('yyyy-MM-dd').format(item.dateModified),
                      style: TextStyles.tableBody))),
            ],
            onLongPress: () {
              controller.selectedItem = item;
              itemController.selectedDuct = ductsList.firstWhere(
                  (duct) => duct.id == controller.selectedItem!.ductId);
              Get.to(() => DuctItemsDetails());
            },
            selected: selectedItem == item,
            onSelectChanged: (isSelected) {
              final isAdding = isSelected != null && isSelected;
              Item? select = controller.selectedItem;
              isAdding ? select = item : select = null;
              controller.selectedItem = select;
            },
          ),
        )
        .toList();
  }
}

import 'package:get/get.dart';

import '../../../!!! controllers/projects_controller.dart';

class DialogAddEditSystem {
  ProjectsController controller = Get.find();

  DialogAddEditSystem(BuildContext context) {
    System? system = controller.selectedSystem;
    final GlobalKey<FormState> formKey = GlobalKey<FormState>();
    String enteredName = '';
    String name = system?.name ?? '';
    int qty = 1;

    showCupertinoDialog(
      context: context,
      builder: (BuildContext context) => CupertinoAlertDialog(
        title: Text(
          system == null ? 'Add Unit Name' : 'Update Unit Name',
          style: TextStyles.messageHeader,
        ),
        content: Column(
          children: [
            Row(
              children: [
                Container(
                  width: 70,
                  height: 30,
                  alignment: Alignment.centerLeft,
                  child: Text('Name', style: TextStyles.messageBody),
                ),
                Container(
                  width: 150,
                  height: 30,
                  padding: EdgeInsets.zero,
                  decoration: BoxDecoration(
                      borderRadius: BorderRadius.circular(10),
                      border: Border.all(color: Colors.grey)),
                  child: CupertinoTextField(
                    key: formKey,
                    padding: EdgeInsets.symmetric(horizontal: 16, vertical: 0),
                    onChanged: (value) {
                      enteredName = value;
                    },
                    controller: TextEditingController(text: name),
                    placeholder: 'Add Unit',
                    style: TextStyles.messageBody,
                    decoration: BoxDecoration(color: Colors.transparent),
                  ),
                ),
              ],
            ),
            SizedBox(height: 5),
            Row(
              children: [
                Container(
                  width: 70,
                  height: 30,
                  alignment: Alignment.centerLeft,
                  child: Text(
                    'Quantity',
                    style: TextStyles.messageBody,
                  ),
                ),
                Container(
                  padding: EdgeInsets.zero,
                  width: 50,
                  height: 30,
                  decoration: BoxDecoration(
                      borderRadius: BorderRadius.circular(10),
                      border: Border.all(color: Colors.grey)),
                  alignment: Alignment.centerLeft,
                  child: CupertinoTextField(
                    style: TextStyles.messageBody,
                    padding: EdgeInsets.zero,
                    decoration: BoxDecoration(color: Colors.transparent),
                    textAlign: TextAlign.center,
                    controller: TextEditingController(
                        text: controller.selectedSystem?.quantity.toString() ??
                            '1'),
                    onChanged: (value) {
                      qty = int.tryParse(value) ?? 1;
                    },
                    keyboardType: TextInputType.number,
                  ),
                ),
              ],
            ),
          ],
        ),
        actions: <Widget>[
          CupertinoDialogAction(
            child: Text(
              'Cancel',
              style: TextStyles.messageCancel,
            ),
            onPressed: () async {
              Navigator.of(context).pop();
            },
          ),
          CupertinoDialogAction(
            child: Text(
              system == null ? 'Add' : 'Change',
              style: TextStyles.messageConfirm,
            ),
            onPressed: () async {
              formKey.currentState?.save();
              if (enteredName == '') {
                enteredName = system?.name ?? 'New Unit';
              }

              if (system == null) {
                await controller.addSystem(enteredName, qty);
              } else {
                system.name = enteredName;
                system.quantity = qty;
                controller.updateDBSystem(system);
              }
              Navigator.of(context).pop();
            },
          ),
        ],
      ),
    );
  }
}

                      } else {
                        DialogAddEditSystem(context);
                      }
                    },
                    child: const Text('Create New Unit'))
              ],
            ),
          ],
        );
      }),
      textConfirm: 'Save',
      onConfirm: () {
        if (controller.selectedProject == null ||
            controller.selectedSystem == null) {
          Get.snackbar('Error', 'Please select project and unit to save');

          itemController.selectedItem.id = controller.selectedItem?.id;
          controller.selectedItem = itemController.selectedItem;
          controller.selectedItem!.projectId = controller.selectedProject!.id!;
          controller.selectedItem!.systemId = controller.selectedSystem!.id!;

          Get.back();
          Get.back();
          Get.back();
          Future.delayed(Duration(milliseconds: 500))
              .then((value) => Get.to(DataTableView()));
          viewController.updateView(dataTableViewPages, 2);

class DialogEditMaterial {

  DialogEditMaterial({required this.material}) {
    final GlobalKey<FormState> formKey = GlobalKey<FormState>();
    SelectedSettingsModel settings = projectController.selectedSettings;
    double enteredValue =
        fromUnit(settings.area, toUnit(settings.weight, material.density));

    showCupertinoDialog(
      context: Get.overlayContext!,
      builder: (BuildContext context) {
        return CupertinoAlertDialog(
          title: Text('Change Material Weight/Area',

                  style: TextStyles.messageBody),
              Row(
                children: [
                  SizedBox(
                    width: 100,
                    child: CupertinoTextField(
                      key: formKey,
                      style: TextStyles.messageBody,
                      onChanged: (value) {
                        enteredValue = double.tryParse(value) ??
                            fromUnit(settings.area,
                                toUnit(settings.weight, material.density));
                      },
                      controller: TextEditingController(
                          text: fromUnit(settings.area,
                                  toUnit(settings.weight, material.density))
                              .toPrecision(2)
                              .toString()),
                      decoration: BoxDecoration(
                        border: Border.all(color: CupertinoColors.black),
                        borderRadius: BorderRadius.circular(10),
                      ),
                      placeholder: 'Add material density',
                    ),
                  ),
                  SizedBox(
                      width: 100,
                      child: Text(
                          '${settings.weight.name}/${settings.area.name}',
                          style: TextStyles.messageBody))
                ],
              ),
            ],
          ),
          actions: [
            CupertinoDialogAction(
              child: Text(
                'Cancel',
                style: TextStyles.messageCancel,
              ),
              onPressed: () {
                Get.back();
              },
            ),
            CupertinoDialogAction(
              child: Text('Change', style: TextStyles.messageConfirm),
              onPressed: () {
                formKey.currentState?.save();
                material.density = toUnit(
                    settings.area, fromUnit(settings.weight, enteredValue));
                controller.updateMaterial(material);
                Get.back();
              },
            ),

      },

import 'package:get/get.dart';

class Controller extends GetxController {
  var isFormVisible = false.obs;
  var isEdited = false.obs;

  Rx<PDModel> pdModel = PDModel(
          shape: 'Rectangular',
          material: 'GI',
          projectSettings: ProjectSettings(
              airFlowUnit: 'l/min',
              widthUnit: 'inch',
              heightUnit: 'inch',
              lengthUnit: 'mt',
              velocityUnit: 'm/s',
              frictionUnit: 'Pa/m',
              pressureDropUnit: 'Pa'),
          airFlowNumber: '800',
          heightNumber: '6',
          widthNumber: '10',
          lengthNumber: '5')
      .obs;

  Rx<ItemModel> selectedItem = ItemModel(
          number: '10000',
          setting: SettingModel(
              dimension: 'Air Flow Rate', category: 'Flow Rate', unit: 'cfm'))
      .obs;

  void toggleFormVisibility(ItemModel item) {
    isFormVisible.value = true;
    isEdited.value = false;
    selectedItem.value = item;
  }

  void crossDimUpdate(String value) {
    SettingModel heightSetting = pdModel.value.projectSettings.heightSetting;
    SettingModel widthSetting = pdModel.value.projectSettings.widthSetting;

    pdModel.update((val) {
      val!.projectSettings.heightSetting = SettingModel(
          dimension: heightSetting.dimension,
          category: heightSetting.category,
          unit: value);
      val.height = ItemModel.calculated(
          setting: pdModel.value.projectSettings.heightSetting,
          value: pdModel.value.height.value);
      val.projectSettings.widthSetting = SettingModel(
          dimension: widthSetting.dimension,
          category: widthSetting.category,
          unit: value);
      val.width = ItemModel.calculated(
          setting: pdModel.value.projectSettings.widthSetting,
          value: pdModel.value.width.value);
    });
  }

  void lengthDimUpdate(String value) {
    SettingModel lengthSetting = pdModel.value.projectSettings.lengthSetting;

    pdModel.update((val) {
      val!.projectSettings.lengthSetting = SettingModel(
          dimension: lengthSetting.dimension,
          category: lengthSetting.category,
          unit: value);
      val.length = ItemModel.calculated(
          setting: pdModel.value.projectSettings.lengthSetting,
          value: pdModel.value.length.value);
    });
  }

  void airFlowDimUpdate(String value) {
    SettingModel airflowSetting = pdModel.value.projectSettings.airFlowSetting;

    pdModel.update((val) {
      val!.projectSettings.airFlowSetting = SettingModel(
          dimension: airflowSetting.dimension,
          category: airflowSetting.category,
          unit: value);
      val.airFlow = ItemModel.calculated(
          setting: pdModel.value.projectSettings.airFlowSetting,
          value: pdModel.value.airFlow.value);
    });
  }

  void velocityDimUpdate(String value) {
    SettingModel velocitySetting =
        pdModel.value.projectSettings.velocitySetting;

    pdModel.update((val) {
      val!.projectSettings.velocitySetting = SettingModel(
          dimension: velocitySetting.dimension,
          category: velocitySetting.category,
          unit: value);
      val.velocity = ItemModel.calculated(
          setting: pdModel.value.projectSettings.velocitySetting,
          value: pdModel.value.velocity.value);
    });
  }

  void frictionDimUpdate(String value) {
    SettingModel frictionSetting =
        pdModel.value.projectSettings.frictionSetting;

    pdModel.update((val) {
      val!.projectSettings.frictionSetting = SettingModel(
          dimension: frictionSetting.dimension,
          category: frictionSetting.category,
          unit: value);
      val.friction = ItemModel.calculated(
          setting: pdModel.value.projectSettings.frictionSetting,
          value: pdModel.value.friction.value);
    });
  }

  void pressureDimUpdate(String value) {
    SettingModel pressureDropSetting =
        pdModel.value.projectSettings.pressureDropSetting;

    pdModel.update((val) {
      val!.projectSettings.pressureDropSetting = SettingModel(
          dimension: pressureDropSetting.dimension,
          category: pressureDropSetting.category,
          unit: value);
      val.pressureDrop = ItemModel.calculated(
          setting: pdModel.value.projectSettings.pressureDropSetting,
          value: pdModel.value.pressureDrop.value);
    });
  }

  void onBtnTap(String value) {
    if (value == KeyboardBtn.done) {
      isFormVisible.value = false;
    } else {
      String text = '';

      if (isEdited.value == true) {
        text = selectedItem.value.number;
      }

      text = KeyboardBtn().onBtnTap(text, value);
      updateNumberValue(text);
      updateItem();
      isEdited.value = true;
    }
  }

  void updateAirFlow(String value) {
    PDModel project = pdModel.value;

    pdModel.value = PDModel(
        shape: project.shape,
        material: project.material,
        projectSettings: project.projectSettings,
        airFlowNumber: value,
        heightNumber: project.height.number,
        widthNumber: project.width.number,
        lengthNumber: project.length.number);
  }

  void updateItem() {
    ItemModel item = selectedItem.value;
    PDModel project = pdModel.value;

    if (item.setting.dimension == 'Air Flow Rate') {
      pdModel.value = PDModel(
          shape: project.shape,
          material: project.material,
          projectSettings: project.projectSettings,
          airFlowNumber: item.number,
          heightNumber: project.height.number,
          widthNumber: project.width.number,
          lengthNumber: project.length.number);
    } else if (item.setting.dimension == 'Width') {
      pdModel.value = PDModel(
          shape: project.shape,
          material: project.material,
          projectSettings: project.projectSettings,
          airFlowNumber: project.airFlow.number,
          heightNumber: project.height.number,
          widthNumber: item.number,
          lengthNumber: project.length.number);
    } else if (item.setting.dimension == 'Length') {
      pdModel.value = PDModel(
          shape: project.shape,
          material: project.material,
          projectSettings: project.projectSettings,
          airFlowNumber: project.airFlow.number,
          heightNumber: project.height.number,
          widthNumber: project.width.number,
          lengthNumber: item.number);
    } else if (item.setting.dimension == 'Height') {
      pdModel.value = PDModel(
          shape: project.shape,
          material: project.material,
          projectSettings: project.projectSettings,
          airFlowNumber: project.airFlow.number,
          heightNumber: item.number,
          widthNumber: project.width.number,
          lengthNumber: project.length.number);
    }
  }

  void backSpace() {
    String text = selectedItem.value.number;
    if (text.isNotEmpty) {
      updateNumberValue(text.substring(0, text.length - 1));
    } else {
      updateNumberValue('');
    }
  }

  void updateNumberValue(String value) {
    selectedItem.update((val) {
      val!.number = value;
    });
  }

  void appendDot() {
    String text = selectedItem.value.number;
    if (text.isEmpty) {
      updateNumberValue('0.');
    } else {
      if (text.contains('.')) {
      } else {
        updateNumberValue('$text.');
      }
    }
  }

  void appendValue(String value) {
    String text = selectedItem.value.number;

    if (text.isEmpty || (double.tryParse(text) == 0 && !(text.contains('.')))) {
      updateNumberValue(value);
    } else {
      updateNumberValue(text + value);
    }
  }

  List<String> getDimensionsList(String query) {
    // ignore: avoid_function_literals_in_foreach_calls
    List<String> filteredList = dimensionsList
        .where((item) => item.toLowerCase().contains(query.toLowerCase()))
        .toList();

    return filteredList;
  }
}

import 'package:get/get.dart';

import 'setting_model.dart';

class ItemModel {
  late String number;
  SettingModel setting;
  late double value;

  ItemModel({
    required this.number,
    required this.setting,
  }) {
    value = setting.rate * (double.tryParse(number) ?? 0);
  }

  ItemModel.calculated({required this.setting, required this.value}) {
    double num = value / setting.rate;

    if (num < 10) {
      number = num.toPrecision(3).toString();
    } else if (num < 100) {
      number = num.toPrecision(2).toString();
    } else if (num < 1000) {
      number = num.toPrecision(1).toString();
    } else if (num < 99999999999) {
      number = num.toInt().toString();
    } else {
      number = '-';
    }
  }
}

import 'package:flutter/material.dart';

class KeyboardBtn {
  static const String del = "C";
  static const String back = "<";
  static const String space = " ";
  static const String cancel = "Cancel";
  static const String done = "Done";

  static const String n0 = "0";
  static const String n1 = "1";
  static const String n2 = "2";
  static const String n3 = "3";
  static const String n4 = "4";
  static const String n5 = "5";
  static const String n6 = "6";
  static const String n7 = "7";
  static const String n8 = "8";
  static const String n9 = "9";
  static const String dot = ".";

  static const List<String> buttonValues = [
    n1,
    n2,
    n3,
    del,
    n4,
    n5,
    n6,
    back,
    n7,
    n8,
    n9,
    space,
    space,
    n0,
    dot,
    done
  ];

  Color getBtnColor(value) {
    return [cancel].contains(value)
        ? Colors.redAccent
        : [done].contains(value)
            ? Colors.blueGrey
            : [space].contains(value)
                ? Colors.transparent
                : Colors.white70;
  }

  String backSpace(String text) {
    if (text.isNotEmpty) {
      return text.substring(0, text.length - 1);
    } else {
      return '';
    }
  }

  String appendDot(String text) {
    if (text.isEmpty) {
      return '0.';
    } else {
      if (text.contains('.')) {
        return text;
      } else {
        return '$text.';
      }
    }
  }

  String appendValue(String text, String value) {
    if (text.isEmpty || (double.tryParse(text) == 0 && !(text.contains('.')))) {
      return value;
    } else {
      return text + value;
    }
  }

  String onBtnTap(String text, String value) {
    if (int.tryParse(value) == null) {
      if (value == KeyboardBtn.dot) {
        return appendDot(text);
      }
      if (value == KeyboardBtn.del) {
        return '';
      }
      if (value == KeyboardBtn.back) {
        return backSpace(text);
      } else {
        return text;
      }
    } else {
      return appendValue(text, value);
    }
  }
}

import 'dart:math';

import 'item_model.dart';
import 'project_setting.dart';

class PDModel {
  String shape;
  String material;
  ProjectSettings projectSettings;

  late ItemModel airFlow;
  late ItemModel height;
  late ItemModel width;
  late ItemModel length;
  late double aspectRatio;
  late ItemModel friction;
  late ItemModel velocity;
  late ItemModel pressureDrop;

  PDModel(
      {required this.shape,
      required this.material,
      required this.projectSettings,
      required String airFlowNumber,
      required String heightNumber,
      required String widthNumber,
      required String lengthNumber}) {
    airFlow = ItemModel(
        number: airFlowNumber, setting: projectSettings.airFlowSetting);

    height =
        ItemModel(number: heightNumber, setting: projectSettings.heightSetting);

    width =
        ItemModel(number: widthNumber, setting: projectSettings.widthSetting);

    length =
        ItemModel(number: lengthNumber, setting: projectSettings.lengthSetting);

    aspectRatio = height.value / width.value;

    double equDiam = 1.3 *
        pow((height.value * width.value), 0.625) /
        pow((height.value + width.value), 0.25);

    double velocityV = airFlow.value / (height.value * width.value);
    double frictionV =
        1 / (50 * 1.095) * pow(airFlow.value, 1.9) / pow(equDiam, 5.02);
    double pressureDropV = length.value * frictionV;

    velocity = ItemModel.calculated(
        setting: projectSettings.velocitySetting, value: velocityV);

    friction = ItemModel.calculated(
        setting: projectSettings.frictionSetting, value: frictionV);
    pressureDrop = ItemModel.calculated(
        setting: projectSettings.pressureDropSetting, value: pressureDropV);
  }
}
